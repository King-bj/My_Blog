CREATE PROCEDURE InsertAlarmCompensationData()
BEGIN
    DECLARE n INT DEFAULT 1;
    DECLARE batch_size INT DEFAULT 0;
    DECLARE max_rows INT DEFAULT 500000;  -- 50万条，便于观察99% NULL + 1% 非NULL

    -- 可选：分批提交，避免事务过大
    DECLARE commit_every INT DEFAULT 5000;

    -- 用于随机生成
    DECLARE rand_val DOUBLE;

    START TRANSACTION;

    WHILE n <= max_rows DO
        -- 决定 status：99% 概率为 NULL，1% 为非 NULL
        SET rand_val = RAND();

        INSERT INTO snapshot_alarm_compensation (
            id,
            alarm_body,
            status,
            failed_msg,
            retry_times,
            create_time,
            create_by,
            update_time,
            update_by
        )
        VALUES (
            CONCAT('alarm_', LPAD(n, 6, '0')),
            JSON_OBJECT(
                'eventId', CONCAT('event_', n),
                'level', ELT(1 + FLOOR(RAND() * 3), 'CRITICAL', 'WARNING', 'INFO'),
                'message', CONCAT('Simulated alarm message for event ', n),
                'timestamp', NOW(3)
            ),
            -- status: 99% 为 NULL，1% 为非 NULL
            CASE 
                WHEN rand_val < 0.99 THEN NULL
                ELSE ELT(1 + FLOOR((rand_val - 0.99) / 0.01 * 4), 'success', 'failed', 'partialrunning', 'unmatchedkeywords')
            END,
            -- failed_msg: 仅当 status = 'failed' 且随机命中时设置
            CASE 
                WHEN rand_val >= 0.99 
                     AND ELT(1 + FLOOR((rand_val - 0.99) / 0.01 * 4), 'success', 'failed', 'partialrunning', 'unmatchedkeywords') = 'failed'
                     AND RAND() < 0.7 
                THEN CONCAT('Error occurred in processing event ', n)
                ELSE NULL 
            END,
            -- retry_times: 0~5 随机
            FLOOR(RAND() * 6),
            -- create_time: 近30天内随机
            DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 30 * 24 * 3600) SECOND),
            -- create_by
            CONCAT('user_', FLOOR(RAND() * 100)),
            -- update_time: create_time 之后或之前1小时内
            DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 30 * 24 * 3600 + RAND() * 3600) SECOND),
            -- update_by
            CONCAT('user_', FLOOR(RAND() * 100))
        );

        SET n = n + 1;
        SET batch_size = batch_size + 1;

        -- 每 5000 条提交一次，避免大事务
        IF batch_size >= commit_every THEN
            COMMIT;
            START TRANSACTION;
            SET batch_size = 0;
        END IF;
    END WHILE;

    COMMIT;
    SELECT CONCAT('✅ Successfully inserted ', max_rows, ' records with ~99% status = NULL') AS Result;
END$$
